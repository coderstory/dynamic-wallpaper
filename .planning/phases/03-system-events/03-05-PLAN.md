// Sources/PicCore/State/HoldStatus.swift —— D-12 的数据落点，零渲染
public struct HoldStatus: Equatable, Sendable {
    public let shouldPlay: Bool
    public let reasons: [HoldReason]     // 已按 order 升序
    public var labels: [String]
    public var summary: String?           // 播放中 nil；暂停时中文串，逗号分隔
}
extension HoldReason {
    public var uiLabel: String            // 手动暂停 / 全屏 / 锁屏 / 显示器熄屏 / 系统睡眠 / 电池供电
}

// Sources/PicCore/State/HoldArbiter.swift —— 纯追加两个成员（W10）
@MainActor public final class HoldArbiter {
    public func applyCurrentDecision()          // 起播路径专用：不碰锚点、不触发 seek

}
