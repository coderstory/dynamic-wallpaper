    public struct FullscreenSignals: Equatable, Sendable {
        public var spaceChangedWhileFullyCovering: Bool   // Space 切换与几何满覆盖在同一次重算里同时成立
        public var frontmostAppChangedWhileFullyCovering: Bool

        /// D-02 的落点：几何**不得单独**作为判定依据。

    }
